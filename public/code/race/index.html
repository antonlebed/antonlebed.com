a block, the unit its places share, and the void ties a ring's menu
stages.

QUESTION. door.py proves that a number ring's places over one rational
prime p share one notch, kept as the COUNT V = v_p(L) and read by each
place through its own ladder, and that a block clocked forever keeps
ONE runaway. It does not say WHICH. Past the bends a place's recurrent
price is N^e = p^(ef), a power of p whose exponent is the local degree
ef, so the cheapest candidate is the place of least local degree. Is
the runaway that place, or does the race to the clock seat it, and
where does a tie-break decide? And the two void ties a cubic ring
stages -- 2 = PQ with residue degrees 1 and 2, and 5 = PQ^2 at one
norm across the unramified/ramified seam -- are they this race read at
the void?

THE OBJECTS. door.py's: a place over p with residue field of N = p^f
elements, ramification e, width w, ladder x_1 = 1 < x_2 < ...; m(0) = 1
and m(j) = x_j + 1; a seated place at exponent a has COLUMN c with
m(c) = a after any clock move. Its door is m(V + 1) - a when it is
seated or N - 1 already divides L, and 1 otherwise; a move costs
N^door. The HOLDER of a clock move is the place it moves. A place's
TAIL INDEX is the least j with every gap x_(i+1) - x_i, i >= j, equal
to the tail gap e. The DEBT of a place behind the count is V + 1 - c.
A block walked ALONE has only its own places on the menu: nothing
else opens, so its count never jumps and no other place's unit reaches
its L.

      if ever, at door m(V + 1) >= 2. A place of norm 2 has N - 1 = 1

# rings: the general monic cubic with squarefree discriminant prime to 6

def disc3(a, b, c):
    return (a * a * b * b - 4 * b ** 3 - 4 * a ** 3 * c - 27 * c * c
            + 18 * a * b * c)


def squarefree(n):
    n = abs(n)
    d = 2
    while d * d <= n:
        if n % (d * d) == 0:
            return False
        d += 1
    return True


def cubic(a, b, c, nmax):
    """The places of Q(alpha), alpha^3 + a alpha^2 + b alpha + c = 0, of
    norm up to nmax. The discriminant is squarefree, so Z[alpha] is the
    maximal order and p factors as the cubic does mod p; a ramified p
    divides it once, so p = P Q^2, both of degree 1, Q tame (p >= 5)."""
    D = disc3(a, b, c)
    check(squarefree(D) and gcd(D, 6) == 1,
          f"x^3 + {a}x^2 + {b}x + {c}: discriminant {D}")
    out = []
    for p in DR.primes_to(nmax):
        roots = [x for x in range(p) if (x ** 3 + a * x * x + b * x + c)
                 % p == 0]
        if D % p == 0:
            check(len(roots) == 2, f"a ramified prime {p} off type P Q^2")
            out += [DR.Place(p, 1, 1, 0), DR.Place(p, 1, 2, 0)]
        elif len(roots) == 3:
            out += [DR.unram(p, 1) for _ in range(3)]
        elif len(roots) == 1:
            out.append(DR.unram(p, 1))
            if p * p <= nmax:
                out.append(DR.unram(p, 2))
        elif p ** 3 <= nmax:
            out.append(DR.unram(p, 3))
    return out

    return (q.p, q.f, q.e, q.w)


def ef(q):
    return q.e * q.f


# ---------------------------------------------------------------------------
# the walker: door.py's move, every tie branched, equal states merged

def menu(pls, st, tally, supply):
    """(best price, [move types]), door.py's rmenu with the supply check
    optional: a block walked alone has no supply beyond its places."""
    ms = []
    for i, a in st.a.items():
        q = pls[i]
        r = DR.door_lcm(q, a, st.L)
        tally["read"] += 1
        tally["mismatch"] += r != DR.door_count(q, a, st.L)
        ms.append((q.N ** r, ("c", i, a, r)))
    best = min(p for p, _ in ms) if ms else None
    for q in pls:
        if best is not None and q.N > best:
            break
        if q.idx in st.a:
            continue
        r = DR.door_lcm(q, 0, st.L)
        tally["read"] += 1
        tally["mismatch"] += r != DR.door_count(q, 0, st.L)
        pr = q.N ** r
        ms.append((pr, ("o" if r == 1 and st.L % (q.N - 1) else "u",
                        q.idx, 0, r)))
        if best is None or pr < best:
            best = pr
    if supply:
        check(best < pls[-1].N, "the supply ran out")
    return best, [t for p, t in ms if p == best]


class Path:
    __slots__ = ("st", "log", "ext", "ties", "paid")

    def __init__(self, st):
        self.st, self.log, self.ext, self.ties, self.paid = st, [], [], [], []


def shared_unit(pls, st, t, tally):
    """PR1 at one move: an opening of a place whose residue degree divides
    that of a place already seated over its prime."""
    kind, i, a, r = t
    if a:
        return
    q = pls[i]
    if any(pls[j].p == q.p and pls[j].f % q.f == 0 for j in st.a):
        tally["unit_bad" if kind == "o" else "unit_door2"] += 1


def tree(pls, st0, n, tally, supply=True, cap=20000):
    """Every tie branched, equal states merged (the first path kept):
    the leaves after n moves."""
    front = {st0.key(): Path(st0)}
    for step in range(n):
        nxt = {}
        for path in front.values():
            best, types = menu(pls, path.st, tally, supply)
            for t in types:
                shared_unit(pls, path.st, t, tally)
                n2 = Path(None)
                n2.log, n2.ext = list(path.log), list(path.ext)
                n2.st = DR.rstep(pls, path.st, t, best, n2.log, n2.ext,
                                 tally)
                n2.ties = path.ties + [step] if len(types) > 1 else path.ties
                n2.paid = path.paid + [best]
                k = n2.st.key()
                if k in nxt:
                    tally["merges"] += 1
                else:
                    nxt[k] = n2
        front = nxt
        check(len(front) <= cap, "the branch cap")
    return list(front.values())


def lowest_path(pls, st, n, tally):
    """The walk that always takes door.py's lowest-first member."""
    log, ext = [], []
    for _ in range(n):
        best, types = menu(pls, st, tally, True)
        st = DR.rstep(pls, st, DR.r_low(types), best, log, ext, tally)
    return log

# the race read off a path

def changes(pls, ext):
    """Every change of holder inside a block: (Y, R, count before Y's move,
    R's count after its move, Y's door)."""
    last, out = {}, []
    for b, i, f, a, v0, v1, r in ext:
        if b in last and last[b][0] != i:
            z, vz = last[b]
            out.append((i, z, v0, vz, r))
        last[b] = (i, v1)
    return out


def race_reading(pls, ext, tot):
    """PR2 over one path's changes of holder."""
    least = {}
    for q in pls:
        least[q.p] = min(least.get(q.p, ef(q)), ef(q))
    for y, z, v, vz, r in changes(pls, ext):
        qy, qz = pls[y], pls[z]
        jy, jz = DR.tail_index(qy.lad)[0], DR.tail_index(qz.lad)[0]
        pend = DR.mdepth(qz.lad, v + 1) - DR.mdepth(qz.lad, vz)
        tot["changes"] += 1
        tot["jump"] += v > vz
        if vz >= max(jy, jz):
            tot["past"] += 1
            tot["past_bad"] += not ef(qy) < ef(qz)
            tot["least_bad"] += ef(qz) == least[qz.p]
            tot["past_tie"] += qy.f * r == qz.f * pend
            tot["past_jump"] += v > vz
        else:
            tot["before"] += 1
            tot["before_rise"] += ef(qy) >= ef(qz)


def runaway(pls, log):
    """block -> the set of holders over the last quarter's clock moves."""
    return DR.late_blocks(log)


def lam_brute_tame5(u, b):
    """The exponent of (O/Q^b)^x for O = Z_5[pi], pi^2 = 5u, brute: an
    element c0 + c1 pi, c0 mod 5^ceil(b/2), c1 mod 5^floor(b/2)."""
    m0, m1 = 5 ** ((b + 1) // 2), 5 ** (b // 2)

    def mul(x, y):
        return ((x[0] * y[0] + 5 * u * x[1] * y[1]) % m0,
                (x[0] * y[1] + x[1] * y[0]) % m1)

    def power(x, k):
        r = (1 % m0, 0)
        while k:
            if k & 1:
                r = mul(r, x)
            x = mul(x, x)
            k >>= 1
        return r

    units = [(c0, c1) for c0 in range(m0) if c0 % 5 for c1 in range(m1)]
    one = (1 % m0, 0)
    E = 4 * 5 ** b
    for ell in (2, 5):
        while E % ell == 0 and all(power(x, E // ell) == one for x in units):
            E //= ell
    return E

def block(types):
    pls = sorted((DR.Place(p, f, e, w) for p, f, e, w in types),
                 key=lambda q: (q.N, q.p, q.e))
    for i, q in enumerate(pls):
        q.idx = i
        q.name = f"{q.N}:{q.p},{q.e},{q.f}#{i}"
    return pls


def seed(pls, exps):
    st = DR.RSt()
    for q, a in zip(pls, exps):
        if a:
            st.a[q.idx] = a
            st.L = st.L * q.lam(a) // gcd(st.L, q.lam(a))
    return st

def section_blocks():
    print("S1  BLOCKS ALONE: THE RACE")
    tot, tally = Counter(), Counter()
    nblocks = walks = leaves = 0
    notleast = notleast_notie = 0
    for size in (2, 3):
        for p in (2, 3, 5, 7):
            tp = [t for t in TYPES if t[0] == p]
            for combo in combinations_with_replacement(tp, size):
                if sum(f * e for _, f, e, _ in combo) > 8:
                    continue
                pls = block(combo)
                nblocks += 1
                seeds = [(0,) * size]
                if size == 2:
                    seeds += [(a1, a2) for a1 in (1, 2, 3)
                              for a2 in (1, 2, 3)]
                else:
                    seeds += [(a,) * 3 for a in (1, 2, 3)]
                for sd in seeds:
                    for path in tree(pls, seed(pls, sd), N_B, tally, False):
                        leaves += 1
                        race_reading(pls, path.ext, tot)
                        late = runaway(pls, path.log).get(p, set())
                        if len(late) == 1:
                            (x,) = late
                            mn = min(ef(pls[j]) for j in path.st.a)
                            if ef(pls[x]) > mn:
                                notleast += 1
                                notleast_notie += not path.ties
                    walks += 1
    print(f"  {nblocks} blocks, {walks} trees, {leaves} leaves; doors read "
          f"{tally['read']}, lcm against count mismatches "
          f"{tally['mismatch']}; merges {tally['merges']}")
    print(f"  changes of holder {tot['changes']}: past the tails "
          f"{tot['past']} (at a tie {tot['past_tie']}), not lowering the "
          f"local degree {tot['past_bad']}, displacing a least-degree "
          f"holder {tot['least_bad']}; before the tails {tot['before']}, "
          f"not lowering {tot['before_rise']}; jumps {tot['jump']}")
    print(f"  runaways above the least seated local degree: {notleast}, "
          f"with no tie on the path {notleast_notie}")
    check(tally["mismatch"] == 0, "the door identity in a block")
    check(tot["past_bad"] == 0, "PR2: a change past the tails")
    check(tot["least_bad"] == 0, "PR2: a least-degree holder displaced")
    check(notleast_notie > 0, "PR3: no runaway above the least without "
          "a tie")

def entry_reading(pls, log, ext, tot, seeded=()):
    """PR7 over one path: every change of holder past the tails with no
    jump, sorted by the exit it takes; seeded, the places seated before
    the path's first move."""
    seated, moves, k = set(seeded), [], 0
    for kind, i, r, price, b in log:
        if kind != "o":
            moves.append((ext[k], frozenset(seated), r))
            k += 1
        seated.add(i)
    last = {}
    for (b, i, f, a, v0, v1, r), before, _ in moves:
        if b in last and last[b][0] != i:
            z, vz, rz, zbefore = last[b]
            qy, qz = pls[i], pls[z]
            jy, jz = DR.tail_index(qy.lad)[0], DR.tail_index(qz.lad)[0]
            if vz >= max(jy, jz) and v0 == vz:
                low = rz < DR.tail_index(qz.lad)[1]
                late = i not in zbefore
                tot["entry"] += 1
                tot["entry_low"] += low and not late
                tot["entry_late"] += late and not low
                tot["entry_both"] += low and late
                tot["entry_bad"] += not (low or late)
        last[b] = (i, v1, r, before)


def void_split(pls, tot, tally):
    """The void tree's changes past the tails, at a tie and strict."""
    t = Counter()
    for path in tree(pls, DR.RSt(), N_B, tally, False):
        race_reading(pls, path.ext, t)
        entry_reading(pls, path.log, path.ext, tot)
    tot["past"] += t["past"]
    tot["tie"] += t["past_tie"]
    tot["strict"] += t["past"] - t["past_tie"]
