image.py

Checks: IMAGE: The colour count theorem; IMAGE: A quadratic ring's ideal image is 2^t rule; IMAGE: The lock's rider erases a choice rule; IMAGE: A curve's image is the continuum rule; IMAGE: The gap rule; IMAGE: Greed's point rule.

ties are all followed: a union of orbits of the colour-preserving
permutations, a finite count over an imaginary quadratic ring where
the walk locks, the continuum over a curve over F_2 with a rational
place besides the one removed (the ideal world at limit.py's corner),
and the lock's rider as what erases a choice rather than what adds
one.

QUESTION. limit.py and element.py follow ONE greedy walk to its limit.

imaginary quadratic rings Z[sqrt(-5)] and the maximal order of
Q(sqrt(-23)), where walks lock; and the rings of five curves over F_2 and

element world: at Q(sqrt(-23)) the lock (2) = P2 P2' carries its rider,
both places over 2 running away with a gap the seed chose. Does that

def check(cond, msg):

def multinomial(sigma, counts):
    """sigma! / (prod counts! (sigma - sum counts)!)"""
    rest = sigma - sum(counts)
    check(rest >= 0, "more items than the colour holds")
    out = factorial(sigma) // factorial(rest)
    for k in counts:
        out //= factorial(k)
    return out

class Places:
    """Every affine place of degree <= DCAP of one ring, labelled, with
    its colour; the element world's riders mapped onto the one place
    of their colour."""

    def __init__(self, W):
        self.W = W
        pl = W.R.places(DCAP)
        self.deg = [p[0] for p in pl]
        self.cls = [p[1] for p in pl]
        self.n = len(pl)
        cnt = Counter(zip(self.deg, self.cls))
        for (d, c), k in cnt.items():
            check(k == W.n[d][c], f"{W.name}: the count against the"
                  f" matrix at ({d}, {c})")
        self.sig_dc = cnt
        self.sig_d = Counter(self.deg)
        self.where = {}
        for i in range(self.n):
            self.where.setdefault((self.deg[i], self.cls[i]), []).append(i)


def lam_int(d, a):
    return (2 ** d - 1) * EL.pow2ceil(a) if a else 1


def lam_state(P, st):
    L = 1
    for i, a in enumerate(st):
        if a:
            x = lam_int(P.deg[i], a)
            L = L // gcd(L, x) * x
    return L


def gcd(a, b):
    while b:
        a, b = b, a % b
    return a


def door(P, st, L, i):
    r = 1
    while L % lam_int(P.deg[i], st[i] + r) == 0:
        r += 1
    return r


def labelled_menu(P, st, element, bias=None):
    """The least moves of the labelled engine as successor states. The
    ideal world moves one place to its door at d r; the element world
    moves the core to its door and adds MINREP(-r c) on the places of
    its colours. bias makes one place dearer by a hair, a price that
    reads a label. None when the least cost exceeds DCAP, where a place
    outside the universe could tie."""
    L = lam_state(P, st)
    W = P.W
    best, out = None, set()
    for i in range(P.n):
        d = P.deg[i]
        if best is not None and d > best[0]:
            continue
        r = door(P, st, L, i)
        cost = d * r + (W.m[W.R.neg[W.R.mul(P.cls[i], r)]] if element
                        else 0)
        key = (cost, i == bias)
        if best is not None and key > best:
            continue
        s = list(st)
        s[i] += r
        if element:
            for (rd, rc), k in W.rider((d, P.cls[i]), r):
                (j,) = P.where[(rd, rc)]
                s[j] += k
        s = tuple(s)
        check(lam_state(P, s) != L, "a least move that does not raise")
        if best is None or key < best:
            best, out = key, set()
        out.add(s)
    if best[0] > DCAP:
        return None
    return out


def labelled_shape(P, st, element, by_degree=False):
    T = EL.pow2ceil(max(st))
    c = Counter()
    for i, a in enumerate(st):
        if a:
            col = P.deg[i] if (by_degree or not element) else \
                (P.deg[i], P.cls[i])
            c[(col, a)] += 1
    return (T, tuple(sorted(c.items())))


def orbit_size(P, shape, element, by_degree=False):
    per = {}
    for (col, a), k in shape[1]:
        per.setdefault(col, []).append(k)
    out = 1
    for col, ks in per.items():
        sig = P.sig_d[col] if (by_degree or not element) else P.sig_dc[col]
        out *= multinomial(sig, ks)
    return out


def free_ideal(W, n):
    """limit.py's corner over the ring's supply: shape keys depth by
    depth, every tie followed."""
    S = LM.Sched("corner", LM.ladder_b(2), LM.price_power(1))
    sup = {d: W.supplied[d] for d in range(1, EL.DMAX + 1)}
    front = {LM.State().key(): LM.State()}
    out = []
    for _ in range(n):
        nxt = {}
        for st in front.values():
            for t, _m in LM.menu(S, st, sup)[1]:
                s2 = LM.apply(S, st, t)
                nxt.setdefault(s2.key(), s2)
        front = nxt
        out.append({(s.T, tuple(sorted(Counter(
            {de: k for de, k in s.seat.items() if k}).items())))
            for s in front.values()})
    return out


def free_element(W, n):
    front = {EL.State().key(): EL.State()}
    out = []
    for _ in range(n):
        nxt = {}
        for st in front.values():
            for t in EL.menu(W, st)[1]:
                s2, _rec = EL.apply(W, st, t)
                nxt.setdefault(s2.key(), s2)
        front = nxt
        keys = set()
        for s in front.values():
            c = Counter()
            for cell, v in s.seat.items():
                for e, k in v.items():
                    if k:
                        c[(cell, e)] += k
            keys.add((s.T, tuple(sorted(c.items()))))
        out.append(keys)
    return out


def brute(P, element, bias=None, nmax=12):
    """The labelled states depth by depth, grouped by shape: per depth
    (shapes, groups off their product, groups off the DEGREE multinomial,
    labelled states). Stops before a depth holding more than LABEL_CAP
    states, counting as it builds so the frontier never outgrows it, or
    reaching a menu the universe cannot price."""
    front = {tuple([0] * P.n)}
    rows = []
    for depth in range(1, nmax + 1):
        nxt, over = set(), False
        for st in front:
            m = labelled_menu(P, st, element, bias)
            if m is None or len(nxt | m) > LABEL_CAP:
                over = True
                break
            nxt |= m
        if over:
            break
        front = nxt
        groups = Counter(labelled_shape(P, s, element) for s in front)
        off = sum(k != orbit_size(P, sh, element)
                  for sh, k in groups.items())
        offd = 0
        if element:
            gd = Counter(labelled_shape(P, s, element, True) for s in front)
            offd = sum(k != orbit_size(P, sh, element, True)
                       for sh, k in gd.items())
        rows.append((depth, set(groups), off, offd, len(front)))
    return rows


def section_o(worlds):
    """IP1 and IP2."""
    print("IP1  THE COLOUR COUNT, BRUTE")
    for W in worlds:
        P = Places(W)
        for element in (False, True):
            if (W.name, element) not in ORBIT_CASES:
                continue
            t0 = time.time()
            rows = brute(P, element)
            free = (free_element if element else free_ideal)(W, len(rows))
            for (depth, shapes, off, offd, n), fk in zip(rows, free):
                check(shapes == fk, f"{W.name}: the shapes at depth"
                      f" {depth} part from the identity-free walker's")
                check(off == 0, f"{W.name}: {off} groups off their orbit"
                      f" at depth {depth}")
            world = "element" if element else "ideal"
            print(f"  {W.name:6s} {world:7s} {P.n} places of degree <="
                  f" {DCAP}; depths 1..{rows[-1][0]}: labelled states"
                  f" {[r[4] for r in rows]}, shapes"
                  f" {[len(r[1]) for r in rows]}; every shape the"
                  f" walker's and every group its product"
                  f" ({time.time() - t0:.1f} s)")
            if element:
                DEG_OFF[W.name] = [r[3] for r in rows]
                print(f"  {'':6s} {'':7s} grouped by degree alone, groups"
                      f" off the degree multinomial per depth"
                      f" {DEG_OFF[W.name]}")
    print("IP2  A PRICE THAT READS A LABEL (control)")
    W = worlds[0]
    P = Places(W)
    x = P.where[(1, 0)][0]
    rows = brute(P, False, bias=x)
    off = [r[2] for r in rows]
    check(sum(off) > 0, "control: a label-reading price keeps every orbit")
    print(f"  {W.name}: one degree-1 place priced a hair dearer on ties:"
          f" groups off their product per depth {off}")

def first_openings(W, start, k=10):
    return [d for d in range(start, EL.DMAX + 1) if W.supplied[d]][:k]


def section_c(worlds):
    """IP3: the corner's two shapes and their orbits."""
    print("IP3  THE CURVE CORNER")
    S = LM.Sched("corner", LM.ladder_b(2), LM.price_power(1))
    for W in worlds:
        sup = {d: W.supplied[d] for d in range(1, EL.DMAX + 1)}
        out, census = LM.branches(S, sup, 10, 290)
        run = Counter(LM.shape(st)[0] for st, lg in out)
        want = {1} | ({2} if sup[2] else set())
        check(set(run) == want, f"{W.name}: runaway degrees {set(run)}")
        one = log10(sup[1]) + sum(log10(sup[d])
                                  for d in first_openings(W, 2))
        two = (log10(sup[2]) + sum(log10(sup[d])
                                   for d in first_openings(W, 3))
               if sup[2] else None)
        print(f"  {W.name:6s} sigma(1..6) {[sup[d] for d in range(1, 7)]};"
              f" runaway degrees over {len(out)} branches"
              f" {dict(sorted(run.items()))}; log10 of each shape's orbit"
              f" over its first ten openings: C of degree 1 {one:.2f}"
              + (f", C of degree 2 {two:.2f}" if two is not None else
                 ", no degree-2 shape"))


def section_e(worlds):
    """IP4: the element image over the curves."""
    print("IP4  THE CURVES' ELEMENT IMAGE")
    for W in worlds:
        t0 = time.time()
        reach = {cell for x in range(W.h) for cell, k in W.rep[x]}
        for cell in reach:
            check(W.n[cell[0]][cell[1]] == 1, "a rider colour holding two")
        out, _ = EL.branches(W)
        least, inside, past, prod, ones = None, set(), 0, None, set()
        for st, log in out:
            trans = max([i for i, x in enumerate(log)
                         if x["open"] and x["rider"]] + [-1])
            opened = []
            for i, x in enumerate(log):
                if not x["open"]:
                    continue
                cell = x["t"][0]
                if cell in reach:
                    inside.add(cell)
                if i > trans:
                    past += 1
                    check(cell[1] == 0, f"{W.name}: an opening past"
                          f" the transient not principal")
                    s = W.n[cell[0]][0]
                    if s < 2:
                        ones.add(cell[0])
                    else:
                        least = s if least is None else min(least, s)
                    opened.append(s)
            for cell in {x["t"][0] for x in log if x["open"]} - reach:
                check(st.nseat[cell] == 1, f"{W.name}: a declined"
                      f" sibling of {cell} seated")
            if prod is None:
                prod = sum(log10(s) for s in opened[:10])
        check(all(d <= 2 for d in ones), f"{W.name}: a lone principal"
              f" place chosen at degree {max(ones or [0])}")
        print(f"  {W.name:6s} rider reach {len(reach)} place(s)"
              f" {sorted(reach)}; {len(out)} branches, {past} openings"
              f" past the transient; degrees opened from a single"
              f" principal place {sorted(ones)}, every other opening"
              f" choosing from at least {least}; opened colours inside"
              f" the reach"
              f" {sorted(inside)}; every opened colour outside it one"
              f" seated place at every branch's end; log10 of the first"
              f" ten such choices {prod:.2f} ({time.time() - t0:.1f} s)")

# the number rings


def colour(F, C, pl, element):
    kind = "split" if pl[2] >= 0 else pl[2]
    base = (pl[0], pl[1], kind)
    return base + (C.cls[pl],) if element else base


TALLY = Counter()


class RingWorld:
    def __init__(self, F, element):
        self.F, self.element = F, element
        self.C = ER.Classes(F)
        self.S = ER.Supply(self.C)
        self.name = F.name + (" element" if element else " ideal")

    def menu(self, st, L):
        F = self.F
        if self.element:
            b1, v1 = ER.abstract_menu(self.S, st, L)
            b2, v2 = ER.ring_menu(F, st, L)
            check(b1 == b2 and v1 == set(v2), f"{self.name}: the menu"
                  f" against the ring's least raising elements")
            TALLY["ring menus"] += 1
            return sorted(v1, key=lambda v: sorted(v))
        cost, ties = ML.menu(F, st, L)
        self.starve(st, L)
        return [frozenset({pl: r}.items()) for pl, r in ties]

    def starve(self, st, L):
        F = self.F
        for p in {pl[1] for pl in st if pl[2] >= 0 and st[pl]}:
            P, Q = F.over(p)
            if bool(st.get(P, 0)) != bool(st.get(Q, 0)):
                s, u = (P, Q) if st.get(P, 0) else (Q, P)
                cs = s[0] ** ML.door(F, s, st[s], L)
                cu = u[0] ** ML.door(F, u, 0, L)
                check(cu > cs, f"{self.name}: starvation at {s}, {u}")
                TALLY["starvation"] += 1


def skey(st):
    return tuple(sorted((p, a) for p, a in st.items() if a))


def run_seed(RW, seed):
    """Every branch from seed followed to its lock: per lock-time state,
    (state, lock vehicle, ties on the branch)."""
    F = RW.F
    front = {skey(seed): (dict(seed), F.lam_state(seed), (), 0)}
    finals = {}
    for _ in range(MAXN):
        nxt = {}
        for k, (st, L, hist, t) in front.items():
            if (len(hist) == LOCK_R and len(set(hist)) == 1
                    and not hist[0][1]):
                finals[k] = (st, hist[0][0], t)
                continue
            vs = RW.menu(st, L)
            tie = len(vs) > 1
            for v in vs:
                s2 = dict(st)
                for pl, a in v:
                    s2[pl] = s2.get(pl, 0) + a
                L2 = F.lam_state(s2)
                check(L2 != L, "a move that does not raise lambda")
                h2 = (hist + ((v, tie),))[-LOCK_R:]
                k2 = skey(s2)
                if k2 in nxt:
                    TALLY["merges"] += 1
                nxt.setdefault(k2, (s2, L2, h2, t + tie))
        front = nxt
        if not front:
            break
    check(not front, f"{RW.name}: a branch from {seed} unlocked after"
          f" {MAXN} moves")
    return finals


def limit_of(st, vehicle):
    lim = {p: a for p, a in st.items() if a}
    for pl, a in vehicle:
        lim[pl] = INF
    return tuple(sorted(lim.items()))


def orbit_sum(RW, seed, limits):
    """The colour count at a seed: distinct limits grouped by shape, each
    group against its product of multinomials over the colours refined
    by the seed. A split colour holds its two conjugates; every other
    colour one place."""
    F, C = RW.F, RW.C

    def col(pl):
        return (colour(F, C, pl, RW.element), seed.get(pl, 0))

    groups = Counter()
    for lim in limits:
        d = dict(lim)
        cols = {}
        for pl in d:
            c = col(pl)
            mates = [q for q in (F.conj(pl), pl) if col(q) == c]
            cols[c] = sorted(set(mates))
        shape = tuple(sorted(
            ((c, tuple(sorted(d.get(q, 0) for q in m)))
             for c, m in cols.items()), key=repr))
        groups[shape] += 1
    total = 0
    for shape, k in groups.items():
        size = 1
        for c, ex in shape:
            size *= multinomial(len(ex), list(Counter(ex).values()))
        check(k == size, f"{RW.name}: a shape at seed {seed} holding {k}"
              f" limits against its orbit {size}")
        total += size
    return total, len(groups)


def section_r(F, element):
    """IP5 or IP6 at one ring."""
    RW = RingWorld(F, element)
    t0 = time.time()
    seeds = ER.seeds(F) if element else \
        [{}] + [ideal for n, ideal in F.ideals(SEED_N)]
    rows = Counter()
    erase = []
    for sd in seeds:
        finals = run_seed(RW, sd)
        ts = {t for st, v, t in finals.values()}
        byl = {}
        for st, v, t in finals.values():
            byl.setdefault(limit_of(st, v), []).append((st, v))
        _, nshape = orbit_sum(RW, sd, byl)
        if not element:
            check(len(ts) == 1, f"{RW.name}: branch tie counts {ts}")
            check(nshape == 1, f"{RW.name}: {nshape} shapes at {sd}")
            check(len(byl) == 2 ** min(ts), f"{RW.name}: {len(byl)}"
                  f" limits against 2^{min(ts)} at {sd}")
        for lim, grp in byl.items():
            if len(grp) > 1:
                s1, v1 = grp[0]
                diff = {p for s2, _ in grp[1:] for p in set(s1) | set(s2)
                        if s1.get(p, 0) != s2.get(p, 0)}
                supp = {pl for pl, a in v1}
                check(all(v2 == v1 for _, v2 in grp),
                      "an erasure across two lock vehicles")
                check(diff <= supp, f"{RW.name}: an erasure off the lock"
                      f" support at {sd}")
                check(len(supp) >= 2, f"{RW.name}: an erasure at a lock"
                      f" carrying no rider")
                erase.append((sd, sorted(diff), sorted(v1), len(grp)))
        rows[(min(ts), max(ts), len(finals), len(byl), nshape)] += 1
    print(f"  {RW.name}: {len(seeds)} seeds ({time.time() - t0:.1f} s)")
    print(f"    (least ties, most ties, lock-time states, limits, shapes):"
          f" seeds {sorted(rows.items())}")
    for sd, diff, v, n in erase:
        print(f"    erasure at seed {sd}: {n} states differing at"
              f" {diff}, one limit, the lock vehicle {v}")
    return erase


def section_q():
    print("IP5, IP6  THE QUADRATIC RINGS")
    TALLY.clear()
    er = {}
    for F in (ML.K5, ML.K23):
        section_r(F, False)
    for F in (ML.K5, ML.K23):
        er[F.name] = section_r(F, True)
    print(f"  starvation checks {TALLY['starvation']}; ring menus read"
          f" {TALLY['ring menus']}; branch merges {TALLY['merges']}")
    check(er[ML.K23.name], "no erasure at Q(sqrt(-23))")
    check(not er[ML.K5.name], "an erasure at Z[sqrt(-5)]")


def section_control():
    print("CONTROLS")
    ER.section_control()
    A, B, C = (2, 2, 0), (2, 2, 1), (3, 3, 1)
    v = frozenset({A: 1, B: 1}.items())
    s1, s2, s3 = {A: 5, B: 3, C: 1}, {A: 3, B: 5, C: 1}, {A: 5, B: 3, C: 2}
    check(limit_of(s1, v) == limit_of(s2, v), "control: on the support")
    check(limit_of(s1, v) != limit_of(s3, v), "control: off the support")
    print("  the limit map: two states differing on a lock's support read"
          " one limit, differing off it two")
