def ev(*e):
    return tuple(list(e) + [0] * (R - len(e)))


def dp(pairs):
    """A Dp from (exponent list, coefficient) pairs."""
    out = Counter()
    for e, c in pairs:
        out[ev(*e)] += c
    return Dp(out)


def uni(exps):

# ------------------------------------------------ atoms and factorizations

ATOM_ID, ATOMS, ATOM_VAL = {}, [], []


def aid(f):
    key = frozenset(f.d.items())
    if key not in ATOM_ID:
        ATOM_ID[key] = len(ATOMS)
        ATOMS.append(f)
        ATOM_VAL.append(sum(f.coeffs()))
    return ATOM_ID[key]


PN = {}


def prod_nn(ids):
    """The product of the atoms with these ids, and whether it is
    nonnegative; ids a sorted tuple."""
    if ids not in PN:
        p = prod([ATOMS[i] for i in ids])
        PN[ids] = (p, nonneg(p))
    return PN[ids]


AT = {}


def atomic_block(ids):
    """Nonnegative, and no split into two nonnegative sub-products."""
    if ids in AT:
        return AT[ids]
    ok = prod_nn(ids)[1]
    n = len(ids)
    if ok and n > 1:
        seen = set()
        for r in range(1, n):
            for sub in combinations(range(n), r):
                a = tuple(sorted(ids[i] for i in sub))
                if a in seen:
                    continue
                seen.add(a)
                b = tuple(sorted(ids[i] for i in range(n) if i not in sub))
                if prod_nn(a)[1] and prod_nn(b)[1]:
                    ok = False
                    break
            if not ok:
                break
    AT[ids] = ok
    return ok


def factorizations(ids):
    """The distinct atomic factorizations of the product of the atoms
    with these ids: sorted tuples of blocks, a block a sorted id tuple."""
    out = set()
    for part in set_partitions(list(range(len(ids)))):
        blocks = tuple(sorted(tuple(sorted(ids[i] for i in b))
                              for b in part))
        if all(atomic_block(b) for b in blocks):
            out.add(blocks)
    return sorted(out)


def value(block):
    v = 1
    for i in block:
        v *= ATOM_VAL[i]
    return v


def within(a, b):
    """Multiset inclusion of atom ids: block a divides block b."""
    return not (Counter(a) - Counter(b))


def classify(x, y):
    """SHARED or MIXING, with spectators set aside."""
    cx, cy = Counter(x), Counter(y)
    common = cx & cy
    xs = sorted((cx - common).elements())
    ys = sorted((cy - common).elements())
    shared = any(within(a, b) or within(b, a) for a in xs for b in ys)
    return {
        "verdict": "SHARED" if shared else "MIXING",
        "spectators": sum(common.values()),
        "shape": (tuple(sorted(value(b) for b in xs)),
                  tuple(sorted(value(b) for b in ys))),
        "reducible": all(len(b) > 1 for b in xs + ys),
        "trinomial": any(value(b) == 3 and len(support(b)) == 3
                         for b in xs + ys),
        "x": xs, "y": ys}


def support(block):
    p = prod_nn(block)[0]
    return p.d

def sub(a, b):
    return tuple(i - j for i, j in zip(a, b))


def add(a, b):
    return tuple(i + j for i, j in zip(a, b))


def scale(k, a):
    return tuple(k * i for i in a)


def line_of(points):
    """For collinear points: (primitive direction, least point,
    positions as multiples of the direction), else None."""
    pts = sorted(points)
    diffs = [sub(p, pts[0]) for p in pts[1:]]
    if rank([pts[0]] + list(pts[1:])) != 1:
        return None
    base = next(d for d in diffs if any(d))
    g = 0
    for c in base:
        g = gcd(g, abs(c))
    v = tuple(c // g for c in base)
    pos = []
    for d in diffs:
        k = next(dc // vc for dc, vc in zip(d, v) if vc)
        assert tuple(k * vc for vc in v) == d
        pos.append(k)
    if min(pos) < 0:
        v = scale(-1, v)
        pos = [-k for k in pos]
    return v, pts[0], [0] + pos


def pairs_split(points, steps):
    """Split the points into runs {p, p + t1, ...} along the given step
    lists, in order; return the runs' starting points or None."""
    pts = set(points)
    if not steps:
        return [] if not pts else None
    first = steps[0]
    size = len(first)
    for start in sorted(pts):
        run = {add(start, t) for t in first}
        if len(run) == size and run <= pts:
            rest = pairs_split(pts - run, steps[1:])
            if rest is not None:
                return [start] + rest
    return None


def family_e(x, y):
    """e when {x, y} is the family up to monomials, else None."""
    for side, other in ((x, y), (y, x)):
        vals = sorted(value(b) for b in side)
        if vals != [3, 4] or sorted(value(b) for b in other) != [2, 6]:
            continue
        blocks = side + other
        if any(set(prod_nn(b)[0].coeffs()) != {1} for b in blocks):
            return None
        t = next(b for b in side if value(b) == 3)
        q = next(b for b in side if value(b) == 4)
        bi = next(b for b in other if value(b) == 2)
        s6 = next(b for b in other if value(b) == 6)
        ln = line_of(list(support(t)))
        if ln is None:
            return None
        v, _, pos = ln
        if pos[1] * 2 != pos[2] or pos[1] % 2:
            return None
        e = pos[1] // 2
        ve = scale(e, v)
        zero = scale(0, v)
        qs = pairs_split(list(support(q)),
                         [[zero, scale(3, ve)], [zero, ve]])
        ss = pairs_split(list(support(s6)),
                         [[zero, scale(2, ve), scale(4, ve)],
                          [zero, ve, scale(2, ve)]])
        bs = pairs_split(list(support(bi)), [[zero, scale(3, ve)]])
        if qs is None or ss is None or bs is None:
            return None
        zq, zs = sub(qs[1], qs[0]), sub(ss[1], ss[0])
        if zq != zs or rank([zero, v, zq]) != 2:
            return None
        return e
    return None


# ------------------------------------------------ one-variable multisets

def peel(points, k):
    """The multiset S with points = S + {0, k}, peeled from the least
    point, or None."""
    work = Counter(points)
    out = []
    while work:
        x = min(work)
        for y in (x, x + k):
            if work[y] <= 0:
                return None
            work[y] -= 1
            if work[y] == 0:
                del work[y]
        out.append(x)
    return out


def divides_binomial(points, s):
    """Whether 1 + v^s divides the polynomial whose exponent multiset
    is given (least exponent 0 or more)."""
    top = max(points)
    c = [0] * (top + 1)
    for p in points:
        c[p] += 1
    q = [0] * (top + 1)
    for i in range(top + 1):
        q[i] = c[i] - (q[i - s] if i >= s else 0)
    return all(q[i] == 0 for i in range(max(0, top - s + 1), top + 1))


def criterion(h, s, k):
    """MIXING iff s/g and k/g are odd, both at least 3, and some class
    of h mod g has a nonzero alternating sum."""
    g = gcd(s, k)
    s1, k1 = s // g, k // g
    if s1 % 2 == 0 or k1 % 2 == 0 or s1 == 1 or k1 == 1:
        return "SHARED", None
    sums = [0] * g
    for x in h:
        sums[x % g] += (-1) ** ((x - x % g) // g)
    return ("MIXING" if any(sums) else "SHARED"), sums


def by_division(h, s, hp, k):
    shared = (divides_binomial(hp, s) or divides_binomial(h, k)
              or (k % s == 0 and (k // s) % 2 == 1)
              or (s % k == 0 and (s // k) % 2 == 1))
    return "SHARED" if shared else "MIXING"

def count_and_read(f):
    ids = tuple(sorted(aid(a) for a in zfactors(f)))
    facts = factorizations(ids)
    reads = [classify(x, y) for x, y in combinations(facts, 2)]
    return facts, reads


def section_control():
    section("CONTROL -- the six-term identity and the ten-term pair")
    facts, reads = count_and_read(uni(range(6)))
    control("the six-term identity has 2 atomic factorizations, SHARED",
          len(facts) == 2 and [r["verdict"] for r in reads] == ["SHARED"])
    ten = [0, 2, 3, 4, 5, 6, 7, 8, 9, 11]

    a, b = (2, 16), (2, 8, 32, 128, 512)
    c, d = (2, 64), (2, 8, 16, 32, 128)
    same = (Counter(m * n for m in a for n in b)
            == Counter(m * n for m in c for n in d))

    mf = factorizations(ids)
    mr = [classify(x, y)["verdict"] for x, y in combinations(mf, 2)]
    verdict, sums = criterion([0, 2, 4, 6, 8], 3, 5)
    print(f"  in menu realizations: products agree {same}, {len(mf)} "
          f"factorizations, {mr}; criterion {verdict}, H(-1) = {sums}")
    control("the ten-term pair has 2 factorizations and reads MIXING, as "
          "numbers and as menus, and the criterion reads MIXING",
          len(facts) == 2 and [r["verdict"] for r in reads] == ["MIXING"]
          and same and len(mf) == 2 and mr == ["MIXING"]
          and verdict == "MIXING")


def section_cycle(top=30):
    section(f"CYCLE -- T + {{0, alpha}} peeled by {{0, s}}, "
            f"T and alpha to {top}")
    tried = found = direct = 0
    bad = []
    for a in range(0, top + 1):
        for b in range(a, top + 1):
            t = [0, a, b]
            for alpha in range(1, top + 1):
                pts = t + [x + alpha for x in t]
                for s in sorted({x for x in pts if x > 0}):
                    if s == alpha:
                        continue
                    tried += 1
                    if peel(pts, s) is None:
                        continue
                    found += 1
                    g, h = min(alpha, s), max(alpha, s)
                    ok = (len(set(pts)) == 6 and h == 3 * g
                          and sorted(pts) == [min(pts) + g * i
                                              for i in range(6)])
                    direct += len(set(pts)) == 6
                    if not ok:
                        bad.append((t, alpha, s))
    print(f"  {tried} peels tried, {found} second tilings, {direct} "
          f"direct, {len(bad)} not the six-term identity")
    if bad:
        print(f"  first: {bad[:5]}")
    check("every second tiling is direct and the six-term identity",
          found > 0 and not bad and direct == found)


def family_blocks(e):
    a = dp([((0,), 1), ((2 * e,), 1), ((4 * e,), 1)])
    b = dp([((0,), 1), ((3 * e,), 1), ((0, 1), 1), ((e, 1), 1)])
    c = dp([((0,), 1), ((3 * e,), 1)])
    d = dp([((0,), 1), ((2 * e,), 1), ((4 * e,), 1),
            ((0, 1), 1), ((e, 1), 1), ((2 * e, 1), 1)])
    return a, b, c, d


FAMILY_MENUS = [((2, 8, 32), (2, 16, 6, 12), 1),

def section_family():
    section("FAMILY -- e = 1..4 with z a second variable, and in menus")
    ok = True
    for e in range(1, 5):
        a, b, c, d = family_blocks(e)
        f = a * b
        facts, reads = count_and_read(f)
        es = [family_e(r["x"], r["y"]) for r in reads]
        row = (f == c * d and len(f.d) == 12 and set(f.coeffs()) == {1}
               and len(facts) == 2
               and [r["verdict"] for r in reads] == ["MIXING"]
               and reads[0]["spectators"] == 0 and es == [e])
        ok &= row
        print(f"  e = {e}: {len(f.d)} terms, {len(facts)} factorizations,"
              f" {[r['verdict'] for r in reads]}, template e {es}")
    for m1, m2, e in FAMILY_MENUS:
        ids = tuple(sorted(aid(x) for x in menu_factors(m1)
                           + menu_factors(m2)))
        facts = factorizations(ids)
        reads = [classify(x, y) for x, y in combinations(facts, 2)]
        es = [family_e(r["x"], r["y"]) for r in reads]
        terms = Counter(a * b for a in m1 for b in m2)
        row = (len(terms) == 12 and set(terms.values()) == {1}
               and len(facts) == 2
               and [r["verdict"] for r in reads] == ["MIXING"]
               and reads[0]["spectators"] == 0 and es == [e])
        ok &= row
        print(f"  {set(m1)} x {set(m2)}: {len(facts)} factorizations, "
              f"{[r['verdict'] for r in reads]}, template e {es}")
    check("every member has 12 terms, 2 factorizations, MIXING, no "
          "spectator, and the template's e", ok)


def binomial_census(n_h, top):
    rows = {}
    for rest in combinations_with_replacement(range(0, top + 1), n_h - 1):
        h = (0,) + rest
        for s in range(1, top + 1):
            pts = list(h) + [x + s for x in h]
            for k in sorted({x for x in pts if x > 0}):
                if k == s:
                    continue
                hp = peel(pts, k)
                if hp is None:
                    continue
                key = (tuple(sorted(pts)), min(s, k), max(s, k))
                if key in rows:
                    continue
                crit, _ = criterion(h, s, k)
                div = by_division(list(h), s, hp, k)
                rows[key] = (crit, div, len(set(pts)) == len(pts))
    return rows


def section_binomial():
    section("BINOMIAL -- (1 + v^s) H = (1 + v^k) H', H a multiset")
    ok_all = True
    for n_h, top, value_ in ((6, 12, 12), (5, 14, 10), (7, 10, 14)):
        t0 = time.time()
        rows = binomial_census(n_h, top)
        agree = sum(c == d for c, d, _ in rows.values())
        mixing = sum(c == "MIXING" for c, _, _ in rows.values())
        nondirect = sum(not dr for _, _, dr in rows.values())
        mix_nd = sum(c == "MIXING" and not dr
                     for c, _, dr in rows.values())
        print(f"  value {value_}: {len(rows)} tiling pairs ({nondirect} "
              f"with a doubled point), criterion = division at {agree}, "
              f"MIXING {mixing} ({mix_nd} with a doubled point)"
              f"   [{time.time() - t0:.1f} s]")
        ok = agree == len(rows) and len(rows) > 0
        if value_ == 12:
            ok &= mixing == 0 and nondirect > 0
        else:
            ok &= mixing > 0
        ok_all &= ok
    check("the criterion agrees with division everywhere; value twelve "
          "never mixes, doubled points included; ten and fourteen do",
          ok_all)


def section_line():
    section("LINE -- the family with z = v^j, and the split supports")
    a = uni([0, 2, 4])
    b = uni([0, 3, 8, 9])
    f = a * b
    facts, reads = count_and_read(f)
    line_ok = (set(f.coeffs()) == {1} and len(f.d) == 12
               and len(facts) == 2
               and [r["verdict"] for r in reads] == ["MIXING"]
               and sorted(reads[0]["shape"]) == [(2, 6), (3, 4)])
    print(f"  e = 1, j = 8: {len(facts)} factorizations, "
          f"{[(r['verdict'], r['shape']) for r in reads]}")
    split_ok = True
    base = [0, 1, 3, 4, 5, 6, 7, 8, 9, 10, 11, 14]
    for pts in (base, [14 - x for x in base]):
        facts, reads = count_and_read(uni(pts))
        shapes = [r["shape"] for r in reads]
        print(f"  support {sorted(pts)}: {len(facts)} factorizations, "
              f"{[r['verdict'] for r in reads]}, shapes {shapes}")
        split_ok &= (len(facts) == 2
                     and [r["verdict"] for r in reads] == ["MIXING"]
                     and sorted(shapes[0]) == [(2, 2, 3), (3, 4)])
    check("the line member and both split supports read MIXING",
          line_ok and split_ok)


def section_box():
    section("BOX -- three-menus of {2..32} against four-menus of "
            "{2..24}")
    t0 = time.time()
    menus = {3: list(combinations(range(2, 33), 3)),
             4: list(combinations(range(2, 25), 4))}
    ids, seeds = {}, {3: [], 4: []}
    for size, ms in menus.items():
        for m in ms:
            fs = menu_factors(m)
            ids[m] = [aid(f) for f in fs]
            if any(negative(f) for f in fs):
                seeds[size].append(m)
    print(f"  {len(menus[3])} and {len(menus[4])} menus, "
          f"{len(seeds[3])} and {len(seeds[4])} seeds"
          f"   [{time.time() - t0:.1f} s]")
    pairs = {(a, b) for a in seeds[3] for b in menus[4]}
    pairs |= {(a, b) for a in menus[3] for b in seeds[4]}
    found = []
    for a, b in sorted(pairs):
        facts = factorizations(tuple(sorted(ids[a] + ids[b])))
        if len(facts) >= 2:
            found.append((a, b, facts))
        PN.clear()
        AT.clear()
    print(f"  {len(pairs)} pairs walked, {len(found)} collisions"
          f"   [{time.time() - t0:.1f} s]")
    tally = Counter()
    frame_ok = out_ok = lemma_ok = True
    out_mixing = []
    for a, b, facts in found:
        frame = len({m * n for m in a for n in b}) == 12
        f = prod([prod_nn(blk)[0] for blk in facts[0]])
        dim = rank(list(f.d))
        for x, y in combinations(facts, 2):
            r = classify(x, y)
            e = None
            if r["verdict"] == "MIXING":
                e = family_e(r["x"], r["y"])
            key = ("in" if frame else "out", r["verdict"],
                   r["spectators"] > 0, dim, e)
            tally[key] += 1
            if r["verdict"] == "MIXING":
                lemma_ok &= (r["spectators"] == 0 and r["reducible"]
                             and r["trinomial"])
                if frame:
                    frame_ok &= e == 1
                else:
                    out_ok &= dim == 1
                    out_mixing.append((a, b, r["shape"], dim))
            elif frame:
                frame_ok &= r["spectators"] > 0
    n_in = sum(1 for a, b, _ in found
               if len({m * n for m in a for n in b}) == 12)
    print("  frame, verdict, spectator, dimension, family e: pairs")
    for key, n in sorted(tally.items(), key=str):
        print(f"    {key}: {n}")
    for a, b, shape, dim in out_mixing[:12]:
        print(f"    out of frame MIXING {set(a)} x {set(b)} shape {shape}"
              f" dimension {dim}")
    if len(out_mixing) > 12:
        print(f"    ... {len(out_mixing) - 12} more")
    print(f"  {n_in} collisions in frame, {len(found) - n_in} out"
          f"   [{time.time() - t0:.1f} s]")
    check("85253 pairs, 336 factoring two ways, 71 in frame (the earlier "
          "count)",
          len(pairs) == 85253 and len(found) == 336 and n_in == 71)
    check("in frame: shared with a spectator, or MIXING and the family "
          "at e = 1", frame_ok)
    print(f"  out of frame: {len(out_mixing)} MIXING pairs, every one on "
          f"one line: {out_ok}")
    check("every MIXING pair: no spectator, every block reducible, a "
          "trinomial block", lemma_ok)
